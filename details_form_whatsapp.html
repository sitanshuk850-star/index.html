const SHEET_NAME = "Sheet1"; // Google Drive में जिस folder में photos save होंगे, // उसका ID यहां डालें। const FOLDER_ID = "YOUR_GOOGLE_DRIVE_FOLDER_ID"; function doGet() { return HtmlService.createHtmlOutput(`

Form is working

Please open your HTML website.

`); } function doPost(e) { try { const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); const name = e.parameter.name || ""; const address = e.parameter.address || ""; const imageData = e.parameter.image || ""; if (!name || !address || !imageData) { return HtmlService.createHtmlOutput(`

❌ जानकारी पूरी नहीं है

कृपया नाम, पता और फोटो सभी भरें।

`); } // Base64 image को decode करना const parts = imageData.split(","); const contentType = parts[0].match(/data:(.*);base64/)[1]; const bytes = Utilities.base64Decode(parts[1]); const blob = Utilities.newBlob( bytes, contentType, "photo_" + new Date().getTime() + ".jpg" ); // Google Drive folder const folder = DriveApp.getFolderById(FOLDER_ID); const file = folder.createFile(blob); // Image link const imageUrl = file.getUrl(); // Google Sheet में data save sheet.appendRow([ new Date(), name, address, imageUrl ]); return HtmlService.createHtmlOutput(`
✅ Form Successfully Submitted!

आपकी जानकारी सफलतापूर्वक जमा हो गई है।

`); } catch (error) { return HtmlService.createHtmlOutput(`

❌ Error

${error.message}

`); } }
